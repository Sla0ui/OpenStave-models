cmake_minimum_required(VERSION 3.22.1)
project(openstave_engraving LANGUAGES C CXX)

if(NOT VEROVIO_SOURCE_DIR OR NOT SWIG_WRAP_FILE)
    message(FATAL_ERROR "VEROVIO_SOURCE_DIR and SWIG_WRAP_FILE must be passed by Gradle")
endif()

# Verovio's own CMake project builds libverovio.so (and applies the 16 KB page-size linker flag to it).
set(BUILD_AS_ANDROID_LIBRARY ON)
add_subdirectory(${VEROVIO_SOURCE_DIR}/cmake ${CMAKE_CURRENT_BINARY_DIR}/verovio_build)

# Verovio 6.3.0 (upstream develop too) crashes in renderToMIDI on two repeat ends with no repeat start
# between them (A :|| B :||), which scans produce often: ExpansionMap::GenerateExpansionFor keeps its span
# start on measures it already moved into a sub-section, and inserts before a child that is gone. Compile a
# copy in which the measure after a repeat end starts the next span, the usual reading of such repeats; the
# expansion of other input is unchanged. Configuration fails if a Verovio upgrade changes this code.
set(expansion_src ${VEROVIO_SOURCE_DIR}/src/expansionmap.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${expansion_src})
file(READ ${expansion_src} expansion_code)
set(repeat_end_body [=[
                expansion->GetPlistInterface()->AddRefAllowDuplicate(ref);
                expansion->GetPlistInterface()->AddRefAllowDuplicate(ref);
            }
]=])
set(repeat_end_fixed [=[
                expansion->GetPlistInterface()->AddRefAllowDuplicate(ref);
                expansion->GetPlistInterface()->AddRefAllowDuplicate(ref);
                isStartFromPrevious = true;
            }
]=])
# Both repeat-end branches (a repeat end on the left and on the right of a measure) share this body.
string(REPLACE "${repeat_end_body}" "" without_body "${expansion_code}")
string(LENGTH "${expansion_code}" code_length)
string(LENGTH "${without_body}" stripped_length)
string(LENGTH "${repeat_end_body}" body_length)
math(EXPR matches "(${code_length} - ${stripped_length}) / ${body_length}")
if(NOT matches EQUAL 2)
    message(FATAL_ERROR "Verovio's expansionmap.cpp changed: review the repeat-expansion fix in ${CMAKE_CURRENT_LIST_FILE}")
endif()
string(REPLACE "${repeat_end_body}" "${repeat_end_fixed}" fixed_code "${expansion_code}")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/expansionmap.cpp.new "${fixed_code}")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/expansionmap.cpp.new ${CMAKE_CURRENT_BINARY_DIR}/verovio_fixes/expansionmap.cpp COPYONLY)
get_target_property(verovio_sources verovio SOURCES)
list(LENGTH verovio_sources source_count)
list(FILTER verovio_sources EXCLUDE REGEX "/src/expansionmap\\.cpp$")
list(LENGTH verovio_sources remaining_count)
math(EXPR removed "${source_count} - ${remaining_count}")
if(NOT removed EQUAL 1)
    message(FATAL_ERROR "expansionmap.cpp not found exactly once in the verovio target's sources")
endif()
list(APPEND verovio_sources ${CMAKE_CURRENT_BINARY_DIR}/verovio_fixes/expansionmap.cpp)
set_property(TARGET verovio PROPERTY SOURCES ${verovio_sources})

# JNI glue generated by SWIG from bindings/java/verovio.i.
# Verovio sets its include paths per directory (not on the target), so mirror them from cmake/CMakeLists.txt.
add_library(verovio-jni SHARED ${SWIG_WRAP_FILE})
target_include_directories(verovio-jni PRIVATE
        ${VEROVIO_SOURCE_DIR}/include
        ${VEROVIO_SOURCE_DIR}/include/crc
        ${VEROVIO_SOURCE_DIR}/include/midi
        ${VEROVIO_SOURCE_DIR}/include/hum
        ${VEROVIO_SOURCE_DIR}/include/json
        ${VEROVIO_SOURCE_DIR}/include/pugi
        ${VEROVIO_SOURCE_DIR}/include/tuning-library
        ${VEROVIO_SOURCE_DIR}/include/zip
        ${VEROVIO_SOURCE_DIR}/include/vrv
        ${VEROVIO_SOURCE_DIR}/libmei/dist
        ${VEROVIO_SOURCE_DIR}/libmei/addons)
target_link_options(verovio-jni PRIVATE "-Wl,-z,max-page-size=16384")

find_library(log-lib log)
target_link_libraries(verovio-jni verovio ${log-lib})
